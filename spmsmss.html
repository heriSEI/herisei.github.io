function drawStockPartChart(data) { const stockData = data.filter(r => (r["USAGE STATUS"] || "").trim().toLowerCase() === "stock"); const partQty = {}; stockData.forEach(r => { const partName = (r["PART"] || "UNKNOWN").trim() || "UNKNOWN"; partQty[partName] = (partQty[partName] || 0) + (Number(r["QTY"]) || 0); }); const sorted = Object.entries(partQty).sort((a, b) => b[1] - a[1]); const labels = sorted.map(i => i[0]); const values = sorted.map(i => i[1]); const maxValue = Math.max(...values) || 1; // Skala proporsional: tinggi batang maksimal 100px const scaledValues = values.map(v => Math.max(1, (v / maxValue) * 100)); const greenColors = values.map(v => { return `rgba(0, 208, 132, ${0.45 + ((v / maxValue) * 0.55)})`; }); if (chartStockPart) chartStockPart.destroy(); chartStockPart = new Chart(document.getElementById("chartStockPart"), { type: "bar", data: { labels: labels, datasets: [{ label: "Stock QTY", data: scaledValues, backgroundColor: greenColors, borderRadius: 4 }] }, options: { responsive: true, maintainAspectRatio: false, layout: { padding: { top: 8, bottom: 8 } }, plugins: { legend: { display: false }, tooltip: { backgroundColor: "rgba(26, 31, 46, 0.95)", titleColor: "#00d4ff", bodyColor: "#ffffff", borderColor: "#00d084", borderWidth: 1, padding: 10, displayColors: false, callbacks: { label: function(ctx) { const index = ctx.dataIndex; const originalValue = values[index]; return `Stock QTY: ${originalValue}`; } } } }, scales: { y: { display: false, beginAtZero: true, max: 100 }, x: { ticks: { color: "#8892a6", maxRotation: 60, minRotation: 0, font: { size: 11 } }, grid: { display: false } } } } }); }
